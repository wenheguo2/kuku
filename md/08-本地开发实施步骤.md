### 3.1 脚本1：故事整曲MP3合并 (v6)

**目标**：将每个故事目录下的分段 MP3 合并为一个 `full.mp3`

**输入**：`production/audio/{学科}/{分类}/{故事名}/sp_00001.mp3, sp_00002.mp3, ...`  
**输出**：`production/audio/{学科}/{分类}/{故事名}/full.mp3`

**脚本**：`脚本/01_merge_story_mp3.py`（完整代码见脚本文件）

**v6 核心特性**：
- 按 `segments.json` 的 `seq` 字段排序（用 `segment.id` 匹配音频文件）
- **逐段WPM区间归一化**：180-210不改，>210降速到210，<180加速到180
- **逐段 loudnorm 归一化**：EBU R128 (I=-16 LUFS, TP=-1.5, LRA=11)
- 输出 96kbps CBR 24kHz mono
- 完整性校验：缺段 → 跳过并记录
- segments.json 找不到 → 跳过并报错（不 fallback）
- 未匹配MP3 → 告警跳过（不追加末尾）
- ffprobe 超时 → 跳过报错（不返回0.0）
- 2线程并行，启动清理 `_tmp_merge`

```python
"""
将歌曲 MP3 + TXT 传到远端 4080 机器
- 源: production/generated_stories/原创儿歌/
- 目标: 远端 D:\lrc_work\songs\ (保持相对目录结构)

import os
import sys

REMOTE_USER = os.environ["REMOTE_USER"]

LOCAL_SONG_ROOT = Path("production/generated_stories/原创儿歌")

def main():

    for root, dirs, files in os.walk(LOCAL_SONG_ROOT):

                rel_path = local_path.relative_to(LOCAL_SONG_ROOT)

    for i, (local_path, remote_dir, filename) in enumerate(files_to_transfer):
```

```python
将远端生成的 LRC 文件传回本机 production/ 目录

import paramiko
from pathlib import Path
import os

REMOTE_HOST = os.environ["REMOTE_HOST"]  # 见本地 .env，不入库
REMOTE_USER = os.environ["REMOTE_USER"]
REMOTE_PASS = os.environ["REMOTE_PASS"]

def main():
    ssh = paramiko.SSHClient()
    ssh.set_missing_host_key_policy(paramiko.AutoAddPolicy())
    ssh.connect(REMOTE_HOST, username=REMOTE_USER, password=REMOTE_PASS, timeout=10)
    sftp = ssh.open_sftp()

    # 递归遍历远端 output 目录
    def fetch_dir(remote_dir, local_dir):
        local_dir.mkdir(parents=True, exist_ok=True)
        count = 0
        for item in sftp.listdir_attr(remote_dir):
            remote_path = f"{remote_dir}/{item.filename}"
            local_path = local_dir / item.filename
            if str(item).startswith("d"):  # directory
                count += fetch_dir(remote_path, local_path)
            elif item.filename.endswith(".lrc"):
                sftp.get(remote_path, str(local_path))
                count += 1
        return count

    total = fetch_dir(REMOTE_OUTPUT, LOCAL_SONG_ROOT)
    print(f"Fetched {total} LRC files")

    sftp.close()
    ssh.close()

if __name__ == "__main__":
```

**脚本**：`脚本/03_generate_lrc.py`

```python
"""
歌曲LRC生成脚本（Whisper强制对齐）
- 遍历 D:\lrc_work\songs\ 下所有 .mp3
- 找到同名的 .txt 歌词文件
- 用 Whisper 进行强制对齐（forced alignment）
- 生成 .lrc 文件到 D:\lrc_work\output\
- 使用 Whisper medium 模型（中文效果好），GPU加速
- ⚠️ 在远端 4080 机器上运行: ssh $REMOTE_USER@$REMOTE_HOST（凭据见本地 .env，不入库）
"""
import os
import json
import whisper
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor, as_completed

# 远端路径
SONG_ROOT = Path("D:/lrc_work/songs")
OUTPUT_ROOT = Path("D:/lrc_work/output")
LOG_FILE = Path("D:/lrc_work/logs/03_lrc_generation.log")
FAIL_FILE = Path("D:/lrc_work/logs/03_lrc_failed.json")

# Whisper模型（首次运行会自动下载）
MODEL_NAME = "medium"  # medium 平衡速度和精度；large-v3 精度最高但慢
model = None

def get_model():
    global model
    if model is None:
        print(f"Loading Whisper {MODEL_NAME} model...")
        model = whisper.load_model(MODEL_NAME)
        print("Model loaded.")
    return model

def find_songs():
    """找到所有有对应 .txt 的 .mp3 文件"""
    songs = []
    for root, dirs, files in os.walk(SONG_ROOT):
        for f in files:
            if f.endswith(".mp3"):
                mp3_path = Path(root) / f
                txt_path = mp3_path.with_suffix(".txt")
                # LRC 输出到 output 目录（保持相对结构）
                rel_path = mp3_path.relative_to(SONG_ROOT)
                lrc_path = OUTPUT_ROOT / rel_path.parent / f"{mp3_path.stem}.lrc"
                if txt_path.exists() and not lrc_path.exists():
                    songs.append((mp3_path, txt_path, lrc_path))
    return songs

def generate_lrc(mp3_path, txt_path, lrc_path):
    """用Whisper强制对齐生成LRC"""
    try:
        # 读取歌词文本
        lyrics = txt_path.read_text(encoding="utf-8").strip().split("\n")
        lyrics = [line.strip() for line in lyrics if line.strip()]

        if not lyrics:
            return ("skip", str(mp3_path), "empty lyrics")

        # Whisper 转录（带时间戳）
        m = get_model()
        result = m.transcribe(str(mp3_path), language="zh", word_timestamps=True)

        # 提取 segments 时间戳
        segments = result.get("segments", [])

        # 将 Whisper segments 与歌词行匹配
        # 策略：按顺序匹配，每个 Whisper segment 对应一行歌词
        lrc_lines = [f"[ti:{mp3_path.stem}]"]
        for i, seg in enumerate(segments):
            if i < len(lyrics):
                start_sec = seg["start"]
                mm = int(start_sec // 60)
                ss = start_sec % 60
                lrc_lines.append(f"[{mm:02d}:{ss:05.2f}]{lyrics[i]}")

        # 确保 LRC 输出目录存在
        lrc_path.parent.mkdir(parents=True, exist_ok=True)
        # 写入 LRC 文件
        lrc_path.write_text("\n".join(lrc_lines), encoding="utf-8")
        return ("ok", str(mp3_path), f"{len(lrc_lines)-1} lines")

    except Exception as e:
        return ("fail", str(mp3_path), str(e)[:200])

def main():
    LOG_FILE.parent.mkdir(parents=True, exist_ok=True)
    songs = find_songs()
    print(f"Found {len(songs)} songs to process")

    # 预加载模型
```

### 3.4 脚本4：立绘去白底透明化（rembg）

**目标**：将 1,668 张白底 PNG 去背景，输出透明底 PNG

**输入**：`production/illustrations/characters_generated/{角色}/{pose}/{表情}/xxx.png`  
**输出**：`production/illustrations/characters_transparent/{角色}/{pose}/{表情}/xxx.png`

**脚本**：`脚本/04_rembg_transparent.py`

```python
"""
立绘去白底透明化脚本（rembg）
- 遍历 production/illustrations/characters_generated/ 下所有 PNG
- 用 rembg 去除白色背景（本机 CPU 模式，onnxruntime）
- 输出到 production/illustrations/characters_transparent/ 保持相同目录结构
"""
import os
from rembg import remove, new_session
from pathlib import Path
from PIL import Image
import io
from concurrent.futures import ThreadPoolExecutor, as_completed

INPUT_ROOT = Path("production/illustrations/characters_generated")
OUTPUT_ROOT = Path("production/illustrations/characters_transparent")
LOG_FILE = Path("脚本/logs/04_rembg.log")
FAIL_FILE = Path("脚本/logs/04_rembg_failed.json")

session = None

def get_session():
    global session
    if session is None:
        print("Initializing rembg session (u2net model)...")
        session = new_session("u2net")
        print("Session ready.")
    return session

def find_images():
    """找到所有待处理的 PNG"""
    images = []
    for root, dirs, files in os.walk(INPUT_ROOT):
        for f in files:
            if f.endswith(".png"):
                src = Path(root) / f
                rel = src.relative_to(INPUT_ROOT)
                dst = OUTPUT_ROOT / rel
                if not dst.exists():
                    images.append((src, dst))
    return images

def process_one(src_path, dst_path):
    """处理一张图片"""
    try:
        dst_path.parent.mkdir(parents=True, exist_ok=True)

        # 读取原图
        input_image = Image.open(src_path)

        # rembg 去背景 (remove()直接返回PIL Image)
        output_image = remove(input_image, session=get_session())

        # 保存透明底 PNG
        output_image.save(dst_path, "PNG")

        return ("ok", str(src_path), f"{dst_path.relative_to(OUTPUT_ROOT)}")
    except Exception as e:
        return ("fail", str(src_path), str(e)[:200])

def main():
    LOG_FILE.parent.mkdir(parents=True, exist_ok=True)
    images = find_images()
    print(f"Found {len(images)} images to process")

    get_session()

    results = {"ok": 0, "fail": 0}
    failures = []

    # rembg 可以多线程，但 GPU 显存有限，用2线程
    with ThreadPoolExecutor(max_workers=2) as executor:
        futures = {executor.submit(process_one, s, d): s for s, d in images}
        for i, future in enumerate(as_completed(futures)):
            status, path, msg = future.result()
            results[status] += 1
            if status == "fail":
                failures.append({"path": path, "error": msg})
            if (i + 1) % 100 == 0:
                print(f"  Progress: {i+1}/{len(images)} | ok={results['ok']} fail={results['fail']}")

    with open(LOG_FILE, "w", encoding="utf-8") as f:
        f.write(f"Total: {len(images)}\n")
        for k, v in results.items():
            f.write(f"  {k}: {v}\n")

    with open(FAIL_FILE, "w", encoding="utf-8") as f:
        import json
        json.dump(failures, f, ensure_ascii=False, indent=2)

    print(f"\nDone! ok={results['ok']} fail={results['fail']}")
    print(f"Output: {OUTPUT_ROOT}")
    if failures:
        print(f"Failures: {FAIL_FILE}")
        print("建议人工检查失败图片，用Photoshop手动去背景")

if __name__ == "__main__":
    main()
```

**预估**：1,668张，本机 CPU 模式约 3-5秒/张，约 **1-2 小时**

### 3.5 脚本5：封面图规格化

**目标**：将已有封面图统一为 800×800px WebP（1:1比例，等比缩放不裁剪）

**输入**：`production/illustrations/covers/generated/{学科}/xxx.png`  
**输出**：同目录下 `xxx.webp`

**脚本**：`脚本/05_normalize_covers.py`

```python
"""
封面图规格化脚本
- 遍历 production/illustrations/covers/generated/ 下所有图片
- 统一等比缩放到 800x800px 画布内（不裁剪不截断）
- 输出 WebP 格式（质量85）
- 原图保留不动
"""
from PIL import Image
from pathlib import Path
import os

COVER_ROOT = Path("production/illustrations/covers/generated")
TARGET_W, TARGET_H = 800, 800
QUALITY = 85

def find_images():
    images = []
    for root, dirs, files in os.walk(COVER_ROOT):
        for f in files:
            if f.endswith((".png", ".jpg", ".jpeg", ".webp")) and not f.endswith("_thumb.webp"):
                images.append(Path(root) / f)
    return images

def process_one(src):
    dst = src.with_suffix(".webp")
    if dst.exists() and dst != src:
        return ("skip", str(src))

    try:
        img = Image.open(src).convert("RGB")
        # 等比缩放到目标画布内（contain模式，不裁剪）
        src_w, src_h = img.size
        scale = min(TARGET_W / src_w, TARGET_H / src_h)
        new_w = int(src_w * scale)
        new_h = int(src_h * scale)
        img = img.resize((new_w, new_h), Image.LANCZOS)

        # 创建白色画布，居中放置
        canvas = Image.new("RGB", (TARGET_W, TARGET_H), (255, 255, 255))
        offset_x = (TARGET_W - new_w) // 2
        offset_y = (TARGET_H - new_h) // 2
        canvas.paste(img, (offset_x, offset_y))
        canvas.save(dst, "WEBP", quality=QUALITY)
        return ("ok", str(src))
    except Exception as e:
        return ("fail", str(src), str(e)[:200])

def main():
    images = find_images()
    print(f"Found {len(images)} cover images")
    for img in images:
        status = process_one(img)
        print(f"  {status[0]}: {img.name}")
    print("Done!")

if __name__ == "__main__":
    main()
```

### 3.6 脚本6：全局索引生成

**目标**：生成 `_global.json` 全局索引文件

**脚本**：`脚本/06_generate_global_index.py`

```python
"""
全局索引生成脚本
- 扫描 production/index/generated_stories/ 下各学科的 _index.json
- 汇总生成 _global.json
- 格式：{ subjects: [...], total_stories: N, total_songs: N, total_lessons: N }
"""
import json
from pathlib import Path

INDEX_ROOT = Path("production/index/generated_stories")
OUTPUT_FILE = INDEX_ROOT / "_global.json"

def main():
    subjects = []

    for subj_dir in sorted(INDEX_ROOT.iterdir()):
        if not subj_dir.is_dir():
            continue
        index_file = subj_dir / "_index.json"
        if not index_file.exists():
            continue

        index_data = json.loads(index_file.read_text(encoding="utf-8"))

        subjects.append({
            "id": subj_dir.name,
            "name": index_data.get("name", subj_dir.name),
            "type": index_data.get("type", "story"),  # story / song / lesson
            "categories": index_data.get("categories", []),
            "cover": index_data.get("cover", ""),
            "description": index_data.get("description", "")
        })

    global_index = {
        "version": "1.0",
        "generated_at": "2026-07-14",
        "subjects": subjects,
        "stats": {
            "total_subjects": len(subjects)
        }
    }

    OUTPUT_FILE.write_text(json.dumps(global_index, ensure_ascii=False, indent=2), encoding="utf-8")
    print(f"Generated {OUTPUT_FILE} with {len(subjects)} subjects")

if __name__ == "__main__":
    main()
```
